cmake_minimum_required(VERSION 3.20)
project(quant_ipc VERSION 0.1.0 LANGUAGES CXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "quant_ipc targets Linux; on Windows build inside WSL.")
endif()
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  message(FATAL_ERROR "quant_ipc requires a supported GCC or Clang std::atomic_ref implementation.")
endif()

add_library(quant_ipc src/shm_region.cpp)
target_include_directories(quant_ipc PUBLIC include)
target_compile_features(quant_ipc PUBLIC cxx_std_20)
target_compile_options(quant_ipc PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(quant_ipc PUBLIC rt)

add_executable(ipc_demo examples/ipc_demo.cpp)
target_link_libraries(ipc_demo PRIVATE quant_ipc)
target_compile_options(ipc_demo PRIVATE -Wall -Wextra -Wpedantic -Werror)

add_executable(tick_replay examples/tick_replay.cpp)
target_link_libraries(tick_replay PRIVATE quant_ipc)
target_compile_options(tick_replay PRIVATE -Wall -Wextra -Wpedantic -Werror)

add_executable(strategy_consumer examples/strategy_consumer.cpp)
target_link_libraries(strategy_consumer PRIVATE quant_ipc)
target_compile_options(strategy_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)

include(CTest)
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_executable(ipc_tests tests/ipc_tests.cpp)
  target_link_libraries(ipc_tests PRIVATE quant_ipc Threads::Threads)
  target_compile_options(ipc_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME ipc_tests COMMAND ipc_tests)
  set_tests_properties(ipc_tests PROPERTIES TIMEOUT 30)
  add_executable(replay_support_tests tests/replay_support_tests.cpp)
  target_compile_features(replay_support_tests PRIVATE cxx_std_20)
  target_compile_options(replay_support_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME replay_support_tests COMMAND replay_support_tests)
  set_tests_properties(replay_support_tests PROPERTIES TIMEOUT 10)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME replay_integration COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/replay_integration.py $<TARGET_FILE:tick_replay>)
    set_tests_properties(replay_integration PROPERTIES TIMEOUT 30)
    add_test(NAME strategy_integration COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/strategy_integration.py
      $<TARGET_FILE:tick_replay> $<TARGET_FILE:strategy_consumer>)
    set_tests_properties(strategy_integration PROPERTIES TIMEOUT 30)
    execute_process(COMMAND ${Python3_EXECUTABLE} -c "import pyarrow"
      RESULT_VARIABLE PYARROW_IMPORT_RESULT OUTPUT_QUIET ERROR_QUIET)
    if(PYARROW_IMPORT_RESULT EQUAL 0)
      add_test(NAME replay_conversion COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/replay_conversion.py)
      set_tests_properties(replay_conversion PROPERTIES TIMEOUT 30)
    else()
      message(STATUS "PyArrow unavailable: skipping replay_conversion test")
    endif()
  else()
    message(STATUS "Python3 unavailable: skipping replay integration/conversion tests")
  endif()
endif()
